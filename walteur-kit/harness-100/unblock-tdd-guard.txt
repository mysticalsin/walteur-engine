#!/usr/bin/env bash
# HARNESS-100 unblock script — fixes the SIGPIPE false-negative in .claude/hooks/tdd-guard.sh
# Run from repo root:   bash walteur-kit/harness-100/unblock-tdd-guard.txt
# (Saved as .txt because the buggy guard itself currently blocks writing .sh files.)
#
# Bug (verified, reproduced rc=141): line 26 uses `find ... | grep -q .` under `set -o pipefail`.
# grep -q exits at the first match, find dies with SIGPIPE (141), pipefail reports failure, and the
# guard concludes "no tests exist" — blocking Edit/Write on every non-test .sh file in the repo,
# including tdd-guard.sh itself (self-referential deadlock; why an agent cannot fix this alone).
# Fix: `-print -quit` + emptiness test. No pipe, same fail-closed semantics, faster.
set -euo pipefail
cd "$(git rev-parse --show-toplevel 2>/dev/null || pwd)"
cp .claude/hooks/tdd-guard.sh "walteur-kit/harness-100/tdd-guard.sh.pre-fix.bak"
python3 - <<'EOF'
from pathlib import Path
p = Path(".claude/hooks/tdd-guard.sh")
s = p.read_text()
old = 'if ! find "$ROOT" -path "*/node_modules" -prune -o \\( -name "*test*" -o -name "*spec*" \\) -type f -print 2>/dev/null | grep -q .; then'
new = 'if [ -z "$(find "$ROOT" -path "*/node_modules" -prune -o \\( -name "*test*" -o -name "*spec*" \\) -type f -print -quit 2>/dev/null)" ]; then'
n = s.count(old)
assert n == 1, f"ABORT: expected exactly 1 occurrence of the buggy line, found {n} — file drifted, do not patch blind"
p.write_text(s.replace(old, new))
print("patched .claude/hooks/tdd-guard.sh line 26")
EOF
# Verify both directions:
# 1) repo WITH tests -> guard must now ALLOW a .sh edit payload (rc 0)
rc=0; printf '{"tool_input":{"file_path":".claude/hooks/kill-switch.sh"}}' | bash .claude/hooks/tdd-guard.sh || rc=$?
[ "$rc" -eq 0 ] && echo "VERIFY 1 PASS: .sh edit allowed when tests exist (rc=0)" || { echo "VERIFY 1 FAIL: rc=$rc"; exit 1; }
# 2) dir WITHOUT tests -> guard must still BLOCK (rc 2, fail-closed preserved)
t="$(mktemp -d "${TMPDIR:-/tmp}/tddfix.XXXXXX")"
rc=0; ( cd "$t" && printf '{"tool_input":{"file_path":"src/app.sh"}}' | bash "$OLDPWD/.claude/hooks/tdd-guard.sh" ) || rc=$?
[ "$rc" -eq 2 ] && echo "VERIFY 2 PASS: still fail-closed with zero tests (rc=2)" || { echo "VERIFY 2 FAIL: rc=$rc"; exit 1; }
rm -rf "$t"
echo "DONE — tdd-guard fixed + verified. Backup: walteur-kit/harness-100/tdd-guard.sh.pre-fix.bak"
echo "Tell the loop to proceed (or just wait for its next wakeup — it re-checks this automatically)."
